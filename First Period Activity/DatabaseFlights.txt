--
-- Tablas
--

CREATE TABLE airlines (
    airlineid serial NOT NULL,
    name character varying(50) NOT NULL,
    shortcode character varying(3)
);

CREATE TABLE cities (
    code serial NOT NULL,
    name character varying(15) NOT NULL,
    state character varying(15) NOT NULL,
    country character varying(30) NOT NULL
);

CREATE TABLE airports (
    code serial NOT NULL,
    iata character varying(3) NOT NULL,
    name character varying(100) NOT NULL,
    city integer NOT NULL
);

CREATE TABLE planes (
    plane_id serial NOT NULL,
    brand character varying(30) NOT NULL,
    model character varying(30) NOT NULL,
    range_km integer NOT NULL CHECK (range_km >= 0),
    total_seats integer NOT NULL CHECK (total_seats >= 0)
);

CREATE TABLE flights (
    flight_code character varying(10) NOT NULL,
    source integer NOT NULL,
    destination integer NOT NULL,
    arrival time without time zone NOT NULL,
    departure time without time zone NOT NULL,
    duration integer NOT NULL CHECK (duration >= 0),
    airlineid integer NOT NULL,
    distance integer NOT NULL,
    plane integer NOT NULL
);

CREATE TABLE passengers (
    passportno character varying(10) NOT NULL,
    firstname character varying(20) NOT NULL,
    lastname character varying(20) NOT NULL,
    address character varying(100),
    phone character varying(12)
);

CREATE TABLE tickets (
    ticket_number integer NOT NULL,
    date_of_booking date NOT NULL,
    date_of_travel date NOT NULL,
    date_of_cancellation date,
    passportno character varying(10) NOT NULL,
    flight_code character varying(10) NOT NULL,
    price real CHECK (price >= 0)
);

CREATE SEQUENCE ticket1_ticket_number_seq
    AS integer
    START WITH 1
    INCREMENT BY 1
    NO MINVALUE
    NO MAXVALUE
    CACHE 1;

ALTER SEQUENCE ticket1_ticket_number_seq OWNED BY tickets.ticket_number;

ALTER TABLE ONLY tickets ALTER COLUMN ticket_number SET DEFAULT nextval('ticket1_ticket_number_seq'::regclass);

--
-- Claves primarias
--

ALTER TABLE ONLY airlines
    ADD CONSTRAINT airline_pkey PRIMARY KEY (airlineid);

ALTER TABLE ONLY airports
    ADD CONSTRAINT airport_pkey PRIMARY KEY (code);

ALTER TABLE ONLY cities
    ADD CONSTRAINT city_pkey PRIMARY KEY (code);

ALTER TABLE ONLY planes
    ADD CONSTRAINT plane_pkey PRIMARY KEY (plane_id);

ALTER TABLE ONLY flights
    ADD CONSTRAINT flight_pkey PRIMARY KEY (flight_code);

ALTER TABLE ONLY passengers
    ADD CONSTRAINT passenger_pkey PRIMARY KEY (passportno);

ALTER TABLE ONLY tickets
    ADD CONSTRAINT tickets_pkey PRIMARY KEY (ticket_number);

--
-- Índices
--

CREATE INDEX fki_fk_dest ON flights USING btree (destination);

CREATE INDEX fki_flight_code ON tickets USING btree (flight_code);

CREATE INDEX fki_source ON flights USING btree (source);

--
-- Claves foráneas
--

ALTER TABLE ONLY airports
    ADD CONSTRAINT airport_city_code_fkey FOREIGN KEY (city) REFERENCES cities(code) ON UPDATE CASCADE;

ALTER TABLE ONLY flights
    ADD CONSTRAINT fk_dest FOREIGN KEY (destination) REFERENCES airports(code) NOT VALID;

ALTER TABLE ONLY flights
    ADD CONSTRAINT fk_source FOREIGN KEY (source) REFERENCES airports(code) NOT VALID;

ALTER TABLE ONLY flights
    ADD CONSTRAINT flight_airlineid_fkey FOREIGN KEY (airlineid) REFERENCES airlines(airlineid) ON UPDATE CASCADE;

ALTER TABLE ONLY flights
    ADD CONSTRAINT flight_plane_fkey FOREIGN KEY (plane) REFERENCES planes(plane_id);

ALTER TABLE ONLY tickets
    ADD CONSTRAINT flight_code FOREIGN KEY (flight_code) REFERENCES flights(flight_code) NOT VALID;

ALTER TABLE ONLY tickets
    ADD CONSTRAINT ticket1_passportno_fkey FOREIGN KEY (passportno) REFERENCES passengers(passportno) ON UPDATE CASCADE;

--
-- Unique
--

ALTER TABLE ONLY airports
    ADD CONSTRAINT airport_iata_unique UNIQUE(iata);
