-- ============================================================================
-- FUNCTION 1: Get Passed Subjects for a Student in a Course
-- ============================================================================
-- This function returns all subjects that a student has passed (score >= 5)
-- in a specific course.
--
-- Parameters:
--   p_student_idcard: Student's ID card (VARCHAR)
--   p_course_code: Course code (INTEGER)
--
-- Returns: TABLE with subject codes (INTEGER)
-- ============================================================================

CREATE OR REPLACE FUNCTION subjects_passed_mmr_2526(
    p_student_idcard VARCHAR(8),
    p_course_code INTEGER
)
RETURNS TABLE(subject_code INTEGER) AS $$
BEGIN
    RETURN QUERY
    SELECT DISTINCT sc.subject_id
    FROM _da_vtschool_2526.scores sc
    INNER JOIN _da_vtschool_2526.enrollments e ON sc.enrollment_id = e.code
    WHERE e.student = p_student_idcard
      AND e.course = p_course_code
      AND sc.score >= 5;
END;
$$ LANGUAGE plpgsql;


-- ============================================================================
-- FUNCTION 2: Get Pending (Failed) Subjects for a Student in a Course
-- ============================================================================
-- This function returns all subjects that a student has NOT yet passed
-- in a specific course. This includes:
-- - Subjects with score < 5
-- - Subjects with NULL score
-- - Subjects not yet taken (from year 1 or 2 depending on enrollment history)
--
-- Parameters:
--   p_student_idcard: Student's ID card (VARCHAR)
--   p_course_code: Course code (INTEGER)
--
-- Returns: TABLE with subject codes (INTEGER), subject names (VARCHAR), and year (INTEGER)
-- ============================================================================

CREATE OR REPLACE FUNCTION subjects_pending_mmr_2526(
    p_student_idcard VARCHAR(8),
    p_course_code INTEGER
)
RETURNS TABLE(subject_code INTEGER, subject_name VARCHAR(50), subject_year INTEGER) AS $$
BEGIN
    RETURN QUERY
    SELECT DISTINCT s.code, s.name, s.year
    FROM _da_vtschool_2526.subjects s
    INNER JOIN _da_vtschool_2526.subject_courses sc ON s.code = sc.subject_id
    WHERE sc.course_id = p_course_code
      AND s.code NOT IN (
          -- Exclude subjects already passed
          SELECT DISTINCT scr.subject_id
          FROM _da_vtschool_2526.scores scr
          INNER JOIN _da_vtschool_2526.enrollments e ON scr.enrollment_id = e.code
          WHERE e.student = p_student_idcard
            AND e.course = p_course_code
            AND scr.score >= 5
      )
    ORDER BY s.year, s.code;
END;
$$ LANGUAGE plpgsql;
